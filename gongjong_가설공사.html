/** * 이 코드는 GEXF라는 그래프 파일을 읽어서 화면에 멋진 그래프를 그리는 예제야! * 그래프는 점(노드)과 선(엣지)로 이루어진 그림이야. * 확대/축소 버튼, 원래 크기로 돌아가는 버튼, 그리고 화면에 보이는 이름표(레이블) 수를 조절하는 슬라이더도 있어! */ import Graph from "graphology"; // 그래프를 만들고 관리하는 도구야 import { parse } from "graphology-gexf/browser"; // GEXF 파일을 읽는 도구야 import Sigma from "sigma"; // 그래프를 화면에 예쁘게 그려주는 도구야 export default () => { let renderer: Sigma | null = null; // Sigma를 저장할 변수야. 처음엔 비어 있어! // 인터넷에서 "가설공사/가설공사_gexf.gexf"라는 그래프 파일을 가져오는 거야 fetch("★최종 분석 결과★/가설공사/가설공사_gexf.gexf") // ★수정: arctic.gexf 대신 가설공사_gexf.gexf 파일을 불러와★ .then((res) => res.text()) // 파일을 텍스트로 바꿔줘 .then((gexf) => { // GEXF 텍스트를 그래프 형태로 바꾸는 거야 const graph = parse(Graph, gexf); // HTML에서 필요한 버튼과 공간을 가져오는 거야 const container = document.getElementById("sigma-container") as HTMLElement; // 그래프를 그릴 공간 const zoomInBtn = document.getElementById("zoom-in") as HTMLButtonElement; // 확대 버튼 const zoomOutBtn = document.getElementById("zoom-out") as HTMLButtonElement; // 축소 버튼 const zoomResetBtn = document.getElementById("zoom-reset") as HTMLButtonElement; // 원래 크기로 버튼 const labelsThresholdRange = document.getElementById("labels-threshold") as HTMLInputElement; // 이름표 수 조절 슬라이더 // Sigma를 사용해서 그래프를 화면에 그리는 거야 renderer = new Sigma(graph, container, { minCameraRatio: 0.08, // 얼마나 작게 축소할 수 있는지 maxCameraRatio: 3, // 얼마나 크게 확대할 수 있는지 }); const camera = renderer.getCamera(); // 화면을 확대/축소하는 카메라야 // 확대 버튼을 누르면 그래프를 부드럽게 크게 보여줘 zoomInBtn.addEventListener("click", () => { camera.animatedZoom({ duration: 600 }); // 0.6초 동안 부드럽게 확대 }); // 축소 버튼을 누르면 그래프를 부드럽게 작게 보여줘 zoomOutBtn.addEventListener("click", () => { camera.animatedUnzoom({ duration: 600 }); // 0.6초 동안 부드럽게 축소 }); // 원래 크기 버튼을 누르면 그래프를 처음 크기로 돌려줘 zoomResetBtn.addEventListener("click", () => { camera.animatedReset({ duration: 600 }); // 0.6초 동안 부드럽게 원래 크기로 }); // 슬라이더를 움직이면 화면에 보이는 이름표 수를 바꿀 수 있어 labelsThresholdRange.addEventListener("input", () => { renderer?.setSetting("labelRenderedSizeThreshold", +labelsThresholdRange.value); // 슬라이더 값으로 이름표 수 조절 }); // 슬라이더의 시작 값을 Sigma의 기본 설정 값으로 맞춰줘 labelsThresholdRange.value = renderer.getSetting("labelRenderedSizeThreshold") + ""; }); // 그래프를 지우고 싶을 때 사용하는 함수야 return () => { renderer?.kill(); // Sigma를 정리해서 화면에서 그래프를 지워줘 }; };
